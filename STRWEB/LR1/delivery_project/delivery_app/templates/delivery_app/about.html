Пример кода

Вот как мы рассчитываем время доставки:


function calculateDeliveryTime(distance, traffic) {
    // Базовое время доставки - 30 минут
    let baseTime = 30;
    
    // Добавляем время в зависимости от расстояния
    let distanceTime = distance * 2;
    
    // Учитываем пробки
    let trafficTime = traffic * 5;
    
    // Итоговое время
    return baseTime + distanceTime + trafficTime;
}
